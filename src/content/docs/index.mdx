  <script is:inline>{`
    (function () {
      var input = document.getElementById('home-search-input');
      var box = document.getElementById('home-search-results');
      var timer = null;
      var pfPromise = null;

      function getPagefind() {
        if (!pfPromise) pfPromise = import('/pagefind/pagefind.js');
        return pfPromise;
      }

      function showMessage(msg) {
        box.classList.remove('hidden');
        box.innerHTML = '<p class="px-4 py-3 text-sm text-zinc-500 dark:text-zinc-400">' + msg + '</p>';
      }

      function hideBox() {
        box.classList.add('hidden');
        box.innerHTML = '';
        setSearchingState(false);
      }

      function toPlainText(value) {
        return String(value || '').replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim();
      }

      function escapeHtml(value) {
        return String(value)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#39;');
      }

      function getSnippet(value) {
        var plain = toPlainText(value);
        if (!plain) return '';
        if (plain.length > 80) plain = plain.slice(0, 80).trimEnd() + '...';
        return '.......' + plain + '.......';
      }

        function escapeRegExp(value) {
          return String(value).replace(/[.*+?^$()|[\]\\]/g, '\\$&');
        }

        function highlightText(text, query) {
          var safeText = escapeHtml(text || '');
          var tokens = String(query || '').trim().split(/\s+/).filter(Boolean);
          if (!tokens.length) return safeText;
          var pattern = new RegExp('(' + tokens.map(escapeRegExp).join('|') + ')', 'gi');
          return safeText.replace(pattern, '<mark>$1</mark>');
        }

        function setSearchingState(isSearching) {
          var cardsGrid = document.getElementById('home-cards-grid');
          if (!cardsGrid) return;
          cardsGrid.classList.toggle('invisible', !!isSearching);
          cardsGrid.classList.toggle('pointer-events-none', !!isSearching);
        }

      async function runSearch(q) {
        if (!q || q.length < 2) return hideBox();
        setSearchingState(true);
        showMessage('Searching...');

        try {
          var pagefind = await getPagefind();
          var search = await pagefind.search(q);
          var results = await Promise.all(search.results.slice(0, 5).map(function (r) { return r.data(); }));

          if (!results.length) return showMessage('No results found.');

          box.classList.remove('hidden');
          box.innerHTML = results.map(function (item) {
              var rawTitle = item.meta && item.meta.title ? item.meta.title : item.url;
            var snippet = getSnippet(item.excerpt || item.content || '');
              var title = highlightText(rawTitle, q);
              var highlightedSnippet = highlightText(snippet, q);
            return (
              '<a href="' + item.url + '" class="block px-4 py-3 border-b last:border-b-0 border-zinc-200/80 dark:border-white/10 hover:bg-zinc-50 dark:hover:bg-zinc-800/60 transition-colors">' +
                '<p class="text-sm font-semibold text-zinc-800 dark:text-zinc-100">' + title + '</p>' +
                  (snippet ? '<p class="mt-1 text-xs text-zinc-500 dark:text-zinc-400 whitespace-nowrap overflow-hidden text-ellipsis">' + highlightedSnippet + '</p>' : '') +
              '</a>'
            );
          }).join('');
        } catch (e) {
          showMessage('Search is unavailable right now.');
        }
      }

      input.addEventListener('input', function (e) {
        clearTimeout(timer);
        var q = e.target.value.trim();
        setSearchingState(!!q);
        timer = setTimeout(function () { runSearch(q); }, 180);
      });

      input.addEventListener('focus', function () {
        var q = input.value.trim();
        if (q.length >= 2) {
          setSearchingState(true);
          runSearch(q);
        }
      });

      document.addEventListener('click', function (e) {
        if (!box.contains(e.target) && e.target !== input) hideBox();
      });
    })();
  `}</script>
